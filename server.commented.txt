// quote оборачивает имя в обратные кавычки (`k_firm`); вызывается только для имён из словаря tables
const quote = name => '`' + name + '`';
// Маршрут GET /api/tables: отдаёт браузеру словарь tables в JSON, по нему app.js строит меню, формы и таблицы
// Этот маршрут стоит выше /api/:section, иначе слово tables было бы принято за название раздела

// tableFor находит описание таблицы по части адреса :section (например, /api/firms → firms)
function tableFor(req, res) {
  // Object.hasOwn проверяет, что такой ключ действительно есть в tables (а не унаследован от Object); иначе table = null
  const table = Object.hasOwn(tables, req.params.section) ? tables[req.params.section] : null;
  // Если таблицы нет в списке — отвечаем браузеру статусом 404 и JSON с текстом ошибки
  if (!table) res.status(404).json({ error: 'Таблица не найдена' });
  // Возвращаем описание таблицы (или null, если её нет)
  return table;
}
// keyFor разбирает ключ записи из адреса (:key) в массив чисел; для составного ключа значения идут через запятую
function keyFor(req, table) {
  // Например, адрес /api/payments/5,2 даёт '5,2' → части ['5', '2'] (счёт 5, платёж 2)
  const parts = req.params.key.split(',');
  // Ошибка, если частей не столько, сколько ключевых столбцов, или любая часть — не цифры, или число больше максимума INT
  if (parts.length !== table.keys.length || parts.some(x => !/^\d+$/.test(x) || Number(x) > 2147483647)) {
    // Создаём объект ошибки с понятным текстом
    const error = new Error('Некорректный номер записи');
    // Статус 400 (неверный запрос): по нему обработчик ошибок в конце файла поймёт, что ошибка наша
    error.status = 400;
    // Прерываем функцию; ошибку подхватит catch в маршруте
    throw error;
  }
  // Превращаем строки в числа: ['5', '2'] → [5, 2]
  return parts.map(Number);
}
// valuesFor проверяет данные из тела запроса и возвращает пары [имя поля, значение] для SQL; editing = true при изменении записи
function valuesFor(table, body, editing) {
  // Тело запроса должно быть объектом: не пустым, не массивом
  if (!body || typeof body !== 'object' || Array.isArray(body)) {
    // Создаём ошибку
    const e = new Error('Ожидается объект с данными');
    // Статус 400 — ошибка клиента
    e.status = 400;
    // Выбрасываем ошибку
    throw e;
  }
  // Массив result накапливает пары [имя поля, значение]
  const result = [];
  // Проходим по полям из описания таблицы — лишние поля из запроса игнорируются
  for (const f of table.fields) {
    // Пропускаем ключ, если MySQL создаёт его сама (auto), и вообще любой ключ при изменении записи (ключ менять нельзя)
    if ((table.auto && table.keys.includes(f.name)) || (editing && table.keys.includes(f.name))) continue;
    // Берём присланное значение этого поля из req.body
    let value = body[f.name];
    // Отсутствующее, null и пустая строка считаются «значения нет» → null
    if (value === undefined || value === null || value === '') value = null;
    // У строки убираем пробелы по краям; если после этого пусто — тоже null
    if (typeof value === 'string') value = value.trim() || null;
    // Если поле обязательное, а значения нет — ошибка
    if (f.required && value === null) {
      // Создаём ошибку с названием поля
      const e = new Error('Заполните поле «' + f.label + '»');
      // Статус 400
      e.status = 400;
      // Выбрасываем ошибку
      throw e;
    }
    // Если значение есть — проверяем его
    if (value !== null) {
      // Допускаются только строки и числа (объекты, массивы, true/false — нельзя)
      if (typeof value !== 'string' && typeof value !== 'number') {
        // Создаём ошибку
        const e = new Error('Некорректное значение поля «' + f.label + '»');
        // Статус 400
        e.status = 400;
        // Выбрасываем ошибку
        throw e;
      }
      // Приводим значение к строке, чтобы проверять числа и даты регулярными выражениями
      value = String(value);
      // Ошибка, если нарушено хотя бы одно из условий ниже
      if (
        // Слишком длинное значение (больше max символов)
        (f.max && value.length > f.max) ||
        // Значения нет в списке допустимых (choices)
        (f.choices && !f.choices.includes(value)) ||
        // Целое число: только цифры и не больше максимума INT
        (f.type === 'int' && (!/^\d+$/.test(value) || Number(value) > 2147483647)) ||
        // Десятичное число: до 7 цифр, после точки до 2 цифр
        (f.type === 'decimal' && !/^\d{1,7}(\.\d{1,2})?$/.test(value)) ||
        // Дата строго в формате ГГГГ-ММ-ДД
        (f.type === 'date' && !/^\d{4}-\d{2}-\d{2}$/.test(value))
      ) {
        // Создаём ошибку с названием поля
        const e = new Error('Проверьте поле «' + f.label + '»');
        // Статус 400
        e.status = 400;
        // Выбрасываем ошибку
        throw e;
      }
    }
    // Добавляем проверенную пару [имя поля, значение] в результат
    result.push([f.name, value]);
  }
  // Возвращаем все пары для INSERT или UPDATE
  return result;
}
// where строит условие поиска по ключу: для одного ключа `firm_num` = ?, для составного — `a` = ? AND `b` = ?
// Знаки ? — места для значений, которые db.execute подставит отдельно (это защищает от SQL-инъекций)
const where = table => table.keys.map(k => quote(k) + ' = ?').join(' AND ');
// GET /api/:section — получить все записи таблицы (:section берётся из адреса, например /api/firms)
app.get('/api/:section', async (req, res, next) => {
  // Ищем описание таблицы по адресу
  const t = tableFor(req, res);
  // Если таблицы нет, ответ 404 уже отправлен — выходим
  if (!t) return;
  // try/catch ловит ошибки базы данных
  try {
    // Выполняем SELECT всех строк, сортируем по ключу; db.query используется, потому что значений для подстановки нет
    // [rows] — деструктуризация: из результата берём первый элемент (массив строк)
    const [rows] = await db.query('SELECT * FROM ' + quote(t.table) + ' ORDER BY ' + t.keys.map(quote).join(', '));
    // Отправляем строки браузеру как JSON
    res.json(rows);
  } catch (e) {
    // Передаём ошибку обработчику ошибок в конце файла
    next(e);
  }
});
// GET /api/:section/:key — получить одну запись по ключу
app.get('/api/:section/:key', async (req, res, next) => {
  // Ищем описание таблицы по адресу
  const t = tableFor(req, res);
  // Если таблицы нет, ответ 404 уже отправлен — выходим
  if (!t) return;
  // try/catch ловит ошибки проверки ключа и базы данных
  try {
    // execute выполняет подготовленный запрос: ? заменяются значениями ключа из keyFor, отдельно от текста SQL
    const [rows] = await db.execute('SELECT * FROM ' + quote(t.table) + ' WHERE ' + where(t), keyFor(req, t));
    // Если строк нет — отвечаем 404
    if (!rows.length) return res.status(404).json({ error: 'Запись не найдена' });
    // Отправляем найденную запись (первую и единственную строку) как JSON
    res.json(rows[0]);
  } catch (e) {
    // Передаём ошибку обработчику ошибок
    next(e);
  }
});
// POST /api/:section — проверить данные из формы и добавить запись через INSERT
app.post('/api/:section', async (req, res, next) => {
  // Ищем описание таблицы по адресу
  const t = tableFor(req, res);
  // Если таблицы нет, ответ 404 уже отправлен — выходим
  if (!t) return;
  // try/catch ловит ошибки проверки и базы данных
  try {
    // Проверяем req.body (JSON из формы) и получаем пары [поле, значение]; false — это добавление, не изменение
    const v = valuesFor(t, req.body, false);
    // Выполняем INSERT; [r] — объект с результатом (insertId, affectedRows)
    const [r] = await db.execute(
      // Текст запроса: INSERT INTO `таблица` (`поле1`, `поле2`) VALUES (?, ?) — столько ?, сколько полей
      'INSERT INTO ' + quote(t.table) + ' (' + v.map(x => quote(x[0])).join(', ') + ') VALUES (' + v.map(() => '?').join(', ') + ')',
      // Значения для ? — вторые элементы пар, в том же порядке
      v.map(x => x[1])
    );
    // Статус 201 (создано) и JSON: для таблиц с auto возвращаем новый номер из MySQL, иначе — значения ключа из тела запроса
    res.status(201).json({ message: 'Запись добавлена', id: t.auto ? r.insertId : t.keys.map(k => req.body[k]) });
  } catch (e) {
    // Передаём ошибку обработчику ошибок
    next(e);
  }
});

  if (!t) return;
  // try/catch ловит ошибки проверки и базы данных
  try {
    // Разбираем ключ из адреса в массив чисел
    const key = keyFor(req, t);
    // Проверяем тело запроса; true — это изменение, поэтому ключевые поля пропускаются
    const v = valuesFor(t, req.body, true);
    // Проверяем, что запись существует: SELECT 1 возвращает строку, если запись найдена
    const [found] = await db.execute('SELECT 1 FROM ' + quote(t.table) + ' WHERE ' + where(t), key);
    // Если записи нет — отвечаем 404
    if (!found.length) return res.status(404).json({ error: 'Запись не найдена' });
    // Выполняем UPDATE
    await db.execute(
      // Текст запроса: UPDATE `таблица` SET `поле1` = ?, `поле2` = ? WHERE условие по ключу
      'UPDATE ' + quote(t.table) + ' SET ' + v.map(x => quote(x[0]) + ' = ?').join(', ') + ' WHERE ' + where(t),
      // Значения для ?: сначала новые значения полей, затем значения ключа для WHERE
      [...v.map(x => x[1]), ...key]
    );
    // Сообщаем браузеру об успехе
    res.json({ message: 'Запись изменена' });
  } catch (e) {
    // Передаём ошибку обработчику ошибок
    next(e);
  }
});
// DELETE /api/:section/:key — удалить запись по ключу
app.delete('/api/:section/:key', async (req, res, next) => {
  // Ищем описание таблицы по адресу
  const t = tableFor(req, res);
  // Если таблицы нет, ответ 404 уже отправлен — выходим
  if (!t) return;
  // try/catch ловит ошибки проверки и базы данных
  try {
    // Выполняем DELETE с условием по ключу; [r] — результат с полем affectedRows
    const [r] = await db.execute('DELETE FROM ' + quote(t.table) + ' WHERE ' + where(t), keyFor(req, t));
    // affectedRows = 0 означает, что удалять было нечего — отвечаем 404
    if (!r.affectedRows) return res.status(404).json({ error: 'Запись не найдена' });
    // Сообщаем браузеру об успехе
    res.json({ message: 'Запись удалена' });
  } catch (e) {
    // Передаём ошибку обработчику ошибок (например, если на запись ссылаются другие таблицы)
    next(e);
  }
});
// Обработчик ошибок: Express узнаёт его по четырём параметрам; сюда приходит всё, что передано через next(e)
app.use((err, req, res, next) => {
  // Если у ошибки есть status — это наша ошибка проверки (400), отправляем её текст браузеру
  if (err.status) return res.status(err.status).json({ error: err.message });
  // Код MySQL 1451: на запись ссылаются другие таблицы, удалить нельзя (409 — конфликт)
  if (err.errno === 1451) return res.status(409).json({ error: 'Удаление невозможно: на запись ссылаются другие данные. Сначала измените или удалите связанные записи.' });
  // Код MySQL 1452: указанной связанной записи нет — нарушен внешний ключ
  if (err.errno === 1452) return res.status(400).json({ error: 'Связанная запись не найдена. Обновите страницу и выберите существующее значение.' });
  // Код MySQL 1062: такой ключ или уникальное значение уже есть (например, сотрудник уже руководит другим отделом)
  if (err.errno === 1062) return res.status(409).json({ error: 'Запись с таким ключом уже существует, либо этот сотрудник уже руководит другим отделом.' });
  // Коды 1048, 1264, 1292, 1366, 1406: MySQL не принял данные (NULL в обязательном поле, число вне диапазона, неверная дата и т.п.)
  if ([1048, 1264, 1292, 1366, 1406].includes(err.errno)) return res.status(400).json({ error: 'MySQL не принял данные. Проверьте значения полей.' });
  // Любую другую ошибку записываем в консоль сервера, чтобы клиент не видел технических подробностей
  console.error(err.message);
  // Клиенту отправляем общее сообщение со статусом 503 (сервис недоступен)
  res.status(503).json({ error: 'Ошибка подключения или запроса к MySQL. Проверьте сервер и настройки.' });
});
